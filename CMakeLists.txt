cmake_minimum_required(VERSION 3.16)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(4DCAD LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(FD_BUILD_APP "Build the 4DCAD application (needs OpenGL)" ON)
option(FD_BUILD_TESTS "Build unit tests" ON)

include(FetchContent)
set(FETCHCONTENT_QUIET ON)

# --- dependencies ---------------------------------------------------------------
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)
# stb_image_write (PNG / JPEG output), public domain, pinned
FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG d14d46f6b44456eb88beac58cba833a4135d4c89)
FetchContent_Populate(stb)

# --- core (platform independent: no OpenGL / window) ------------------------------
add_library(fdcore STATIC
  src/primitives.cpp src/scene.cpp src/evaluator.cpp src/geodesic.cpp src/camera.cpp
  src/surface.cpp src/codegen.cpp src/examples.cpp src/library.cpp src/animation.cpp src/space.cpp src/polytope.cpp src/media.cpp)
target_include_directories(fdcore PUBLIC src)
target_include_directories(fdcore PRIVATE ${stb_SOURCE_DIR})
target_link_libraries(fdcore PUBLIC nlohmann_json::nlohmann_json)
if(MINGW)
  target_compile_definitions(fdcore PUBLIC _USE_MATH_DEFINES)
endif()
if(MSVC)
  target_compile_definitions(fdcore PUBLIC _USE_MATH_DEFINES NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(fdcore PUBLIC /utf-8)
endif()

if(FD_BUILD_TESTS)
  enable_testing()
  add_executable(fdtests tests/test_core.cpp)
  target_link_libraries(fdtests PRIVATE fdcore)
  target_compile_definitions(fdtests PRIVATE FD_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  add_test(NAME core COMMAND fdtests)
endif()

if(FD_BUILD_APP)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw URL https://github.com/glfw/glfw/releases/download/3.4/glfw-3.4.zip)
  FetchContent_MakeAvailable(glfw)

  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.6-docking GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(imgui)
  add_library(imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/imgui_demo.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(imgui PUBLIC glfw)

  # --- Vulkan backend (optional): headers + volk (runtime loader, no link-time Vulkan SDK) + glslang (GLSL -> SPIR-V)
  option(FD_VULKAN "Build the Vulkan renderer (falls back to OpenGL at runtime without a Vulkan driver)" ON)
  if(FD_VULKAN)
    FetchContent_Declare(vulkan_headers GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git GIT_TAG v1.3.292 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(vulkan_headers)
    FetchContent_Declare(volk GIT_REPOSITORY https://github.com/zeux/volk.git GIT_TAG 1.3.295 GIT_SHALLOW TRUE)
    FetchContent_Populate(volk)
    add_library(volk_static STATIC ${volk_SOURCE_DIR}/volk.c)
    target_include_directories(volk_static PUBLIC ${volk_SOURCE_DIR})
    target_link_libraries(volk_static PUBLIC Vulkan::Headers ${CMAKE_DL_LIBS})
    target_compile_definitions(volk_static PUBLIC VK_NO_PROTOTYPES)
    set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
    set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
    set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
    set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
    set(ENABLE_CTEST OFF CACHE BOOL "" FORCE)
    set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
    set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glslang GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git GIT_TAG 15.0.0 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glslang)
    target_sources(imgui PRIVATE ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp)
    target_link_libraries(imgui PUBLIC volk_static)
    target_compile_definitions(imgui PUBLIC IMGUI_IMPL_VULKAN_NO_PROTOTYPES)
  endif()

  set(APP_SOURCES src/main.cpp src/app.cpp src/ui.cpp src/ui_toolbar.cpp src/ui_library.cpp src/ui_tree.cpp src/ui_properties.cpp src/ui_viewport.cpp src/ui_windows.cpp src/ui_timeline.cpp src/ui_render.cpp src/render_job.cpp src/render_thread.cpp src/keybinds.cpp src/renderer.cpp src/render_gl.cpp src/gl.cpp src/filedialog.cpp)
  if(WIN32)
    add_executable(4DCAD WIN32 ${APP_SOURCES})
    target_link_libraries(4DCAD PRIVATE comdlg32)
  else()
    add_executable(4DCAD ${APP_SOURCES})
  endif()
  target_link_libraries(4DCAD PRIVATE fdcore imgui glfw)
  if(FD_VULKAN)
    target_sources(4DCAD PRIVATE src/vk_context.cpp src/render_vk.cpp)
    target_compile_definitions(4DCAD PRIVATE FD_HAVE_VULKAN)
    target_link_libraries(4DCAD PRIVATE volk_static glslang glslang-default-resource-limits SPIRV)
  endif()
  if(FD_BUILD_TESTS)
    add_executable(fdapptests tests/test_app.cpp src/app.cpp src/render_job.cpp src/render_thread.cpp src/keybinds.cpp src/renderer.cpp src/render_gl.cpp src/gl.cpp)
    target_link_libraries(fdapptests PRIVATE fdcore glfw)
    add_test(NAME app COMMAND fdapptests)
  endif()
  if(MINGW)
    target_link_options(4DCAD PRIVATE -static -static-libgcc -static-libstdc++ -s)
  endif()
endif()
